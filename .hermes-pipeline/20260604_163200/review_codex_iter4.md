type StepState = VariantProps<typeof stepStyles>['state']

type ProgressStepsProps = {
  steps: { label: string }[]
  current: number
  orientation?: 'horizontal' | 'vertical'
  className?: string
}

export function ProgressSteps({
  steps,
  current,
  orientation = 'horizontal',
  className,
}: ProgressStepsProps) {
  const isHorizontal = orientation === 'horizontal'

  const getStepState = (index: number): StepState => {
    if (index < current) return 'complete'
    if (index === current) return 'current'
    return 'upcoming'
  }

  return (
    <div
      role="list"
      aria-label="Progress"
      className={cn(
        'flex',
        isHorizontal ? 'flex-row items-start gap-4' : 'flex-col gap-3',
        className,
      )}
    >
      {steps.map((step, index) => {
        const state = getStepState(index)
        const isComplete = state === 'complete'
        const isCurrent = state === 'current'

        const stateLabel = isComplete
          ? 'completed'
          : isCurrent
            ? 'current'
            : 'upcoming'

        return (
          <div
            key={index}
            role="listitem"
            aria-current={isCurrent ? 'step' : undefined}
            aria-label={`Step ${index + 1}: ${step.label}, ${stateLabel}`}
            className={cn(
              'flex items-center gap-3',
              isHorizontal && index < steps.length - 1 && 'flex-1',
              !isHorizontal && 'flex-row',
            )}
          >
            <div className="flex items-center gap-3">
              <div className={cn(stepStyles({ state }))}>
                {isComplete ? (
                  <Check className="h-4 w-4" aria-hidden="true" />
                ) : (
                  <span>{index + 1}</span>
                )}
              </div>
              <span
                className={cn(
                  'text-sm font-medium',
                  isCurrent
                    ? 'text-atria-accent'
                    : isComplete
                      ? 'text-atria-ink'
                      : 'text-atria-muted',
                )}
              >
                {step.label}
              </span>
            </div>
            {isHorizontal && index < steps.length - 1 && (
              <div className="mx-2 h-px flex-1 bg-atria-border" />
            )}
          </div>
        )
      })}
    </div>
  )
}
